    package hostedcontrolplane

import (
	"time"

	"github.com/prometheus/client_golang/prometheus"
	metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"
	"k8s.io/apimachinery/pkg/types"
	"sigs.k8s.io/controller-runtime/pkg/metrics"
)

var hcpCreationMilestoneSeconds = prometheus.NewHistogramVec(
	prometheus.HistogramOpts{
		Name:    "hypershift_hcp_creation_milestone_seconds",
		Help:    "Time from HCP creation to key milestones during cluster provisioning",
		Buckets: prometheus.ExponentialBuckets(5, 2, 10),
	},
	[]string{"milestone", "platform"},
)

func init() {
	metrics.Registry.MustRegister(hcpCreationMilestoneSeconds)
}

type milestoneTracker struct {
	recorded map[types.NamespacedName]map[string]bool
}

func newMilestoneTracker() *milestoneTracker {
	return &milestoneTracker{
		recorded: make(map[types.NamespacedName]map[string]bool),
	}
}

func (m *milestoneTracker) recordMilestone(hcpKey types.NamespacedName, milestone, platform string, creationTimestamp metav1.Time) bool {
	if m.recorded[hcpKey] == nil {
		m.recorded[hcpKey] = make(map[string]bool)
	}
	if m.recorded[hcpKey][milestone] {
		return false
	}
	elapsed := time.Since(creationTimestamp.Time).Seconds()
	hcpCreationMilestoneSeconds.WithLabelValues(milestone, platform).Observe(elapsed)
	m.recorded[hcpKey][milestone] = true
	return true
}

func (m *milestoneTracker) reset(hcpKey types.NamespacedName) {
	delete(m.recorded, hcpKey)
}

    func (r *HostedControlPlaneReconciler) recordComponentMilestones(ctx context.Context, hcp *hyperv1.HostedControlPlane) {
	hcpKey := client.ObjectKeyFromObject(hcp)
	platform := string(hcp.Spec.Platform.Type)
	componentsList := &hyperv1.ControlPlaneComponentList{}
	if err := r.Client.List(ctx, componentsList, client.InNamespace(hcp.Namespace)); err != nil {
		return
	}

	milestoneMap := map[string]string{
		"etcd":                "etcd_available",
		"kube-apiserver":      "kas_available",
		"openshift-apiserver": "oapi_available",
	}

	for _, comp := range componentsList.Items {
		milestoneName, ok := milestoneMap[comp.Name]
		if !ok {
			continue
		}
		availableCondition := meta.FindStatusCondition(comp.Status.Conditions, string(hyperv1.ControlPlaneComponentAvailable))
		if availableCondition != nil && availableCondition.Status == metav1.ConditionTrue {
			r.milestones.recordMilestone(hcpKey, milestoneName, platform, hcp.CreationTimestamp)
		}
	}

	if hcp.Status.Ready {
		r.milestones.recordMilestone(hcpKey, "hcp_available", platform, hcp.CreationTimestamp)
	}
}

    func (r *HostedControlPlaneReconciler) update(ctx context.Context, ...) (reconcile.Result, error) {
	// Phase 1: Infrastructure-independent work
	r.reconcileInfrastructure(ctx, ...)     // create LB Services
	// release image lookup (fail-fast)

	r.reconcileInfraIndependentComponents(ctx, ..., intraPassAvailability)
